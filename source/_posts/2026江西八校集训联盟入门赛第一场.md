# 2026江西八校集训联盟入门赛第一场

## A

思路：

题目要求选择两个作为一组，剩余两个作为另一组，求两组之和之间的差值绝对值最小的分配方案

考虑枚举第一组的两个人，复杂度为 $16$ 次枚举，那么剩余的两个人分配到第二组

直接算这两部分和的差值，最后对 $ans$ 取 $min$ 即可

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int a[4];
void solve(){
    int sum = 0;
    for(int i = 0; i < 4; ++ i)
        cin >> a[i], sum += a[i];
    int ans = INF;
    for(int i = 0; i < 4; ++ i) {
        for(int j = 0; j < 4; ++ j) {
            if(i == j)
                continue;
            int A = a[i] + a[j], B = sum - A;
            ans = min(ans, abs(A-B));
        }
    }
    Out(ans);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## B

思路：

题意就是将总共的 $T$ $s$ ，转化为 $H:MM:SS$ 的格式，按照常识的 $1h = 3600s, 1min = 60s$ 进行转换即可

注意分和秒的位置保留两位，实现上将整数转为字符就很好处理了

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int a[4];
void solve(){
    int t;
    cin >> t;
    int h = t / 3600;
    t -= h * 3600;
    int m = t / 60;
    t -= m * 60;
    string out = to_string(h), mm = to_string(m), ss = to_string(t);
    if(mm.size() < 2) {
        mm = "0" + mm;
    }
    if(ss.size() < 2) {
        ss = "0" + ss;
    }
    out = out + ":" + mm + ":" + ss;
    Out(out);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## C

思路：

直接模拟题意

考虑维护 $now$ 表示当前车上有多少人，每一站先下车，再上车，按照题目给的限制更新 $now$ 即可

对于下车的人，对 $now = max(0, now-a[i])$ 就能更新了，表示人数够的时候更新为 $now-a[i]$，不够则取 $0$

对于上车的人，考虑求出需要多少人 $tmp = min(C-now,b[i])$，这个值表示实际上的上车人数，用这个就能求到这一战上了多少人， 留了多少人

在每一站更新车上人数后，对过程中每一站的人取最大就是途中最多的人数

经过 $n$ 站之后的 $now$ 就是到达终点的人数

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18, N = 1e3 + 5;
int n, a[N], b[N], C;
void solve(){
    cin >> n >> C;
    for(int i = 1; i <= n; ++ i)
        cin >> a[i] >> b[i];
    
    int now = 0, A = 0, B = 0;
    for(int i = 1; i <= n; ++ i) {
        now = max(0ll, now - a[i]);
        int take = min(b[i], C - now);
        A += b[i] - take;
        now += take;
        B = max(B, now);
    }

    cout << A << ' ' << B << ' ' << now << '\n';
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## D 

思路：

由于 $n$ 为偶数，且题目要求两两配对，那么一共可以配对出 $n/2$ 对

考虑每一对在什么情况下能够做到差值最小，对于任意一个 $a[i]$，显然是离 $a[i]$ 最近的值，差值最小

即，比他大的第一个，或者比他小的第一个，即 $i$ 只考虑和 $i-1$ 配对还是 $i+1$ 配对

那么可以排序后，对于每个偶数位置，与他最优的方案是前一个奇数的位置

证明：

假设当前偶数位置为 $i$，那么相邻的四个数字可以是 $i-1,i,i+1,i+2$，此时若 $i$ 和 $i+1$ 配对，那么 $i-1$ 和 $i+2$ 配对时对答案的贡献总和是更劣的

将两种方案的贡献和写成式子是： $(a[i] - a[i-1] + a[i+2] - a[i+1]) \sim (a[i+1] - a[i] + a[i+2] - a[i-1])$

将两边化简：$a[i] \sim a[i+1]$，由于 $a[i] < a[i+1]$，则第一种方案，即 $i$ 与 $i-1$ 配对的方案更优

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18, N = 2e5 + 5;
int n, a[N], b[N], C;
void solve(){
    cin >> n;
    for(int i = 1; i <= n; ++ i)
        cin >> a[i];
    sort(a + 1, a + n + 1);
    int ans = 0;
    for(int i = 2; i <= n; i += 2) {
        ans += a[i] - a[i-1];
    }
    Out(ans);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## E

思路：

注意到，对于每个询问 $1 \le x_j \le n$，$n \le 10^6$，在一般空间复杂度 $512MB$ 限制时，$10^6$ 数组完全存的下

所以考虑预处理每个 $a[i]$ 的值，对每次询问 $x$，直接输出 $a[x]$ 即可

而预处理只需要按照题目给出的 $a[i] = (p * a[i-1] + q*a[i-2]) \mod M$ 循环一遍即可

注意乘法可能爆 $int$ 类型

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18, N = 1e6 + 5;
int n, k, a[N], b[N], M, p, q, Q;
void solve(){
    cin >> p >> q >> a[1] >> a[2] >> M >> n;
    for(int i = 3; i <= n; ++ i) {
        a[i] = (1ll * p * a[i-1] + 1ll * q * a[i-2] ) % M;
    }
    cin >> Q;
    for(int i = 1, x; i <= Q; ++ i) {
        cin >> x;
        cout << a[x] << '\n';
    }
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## F 

思路：

又是配对问题，考虑使用 $map$ 维护每个值的出现次数，统计每一种 $(x, y)$ 的配对对答案贡献了多少次

当然，也可以使用离散化将 $10^9$ 的数值，映射到 $10^5$ 的数量级存放到数组中，

不过 $map$ 更方便一点，但也存在被卡常的情况，这里没有

要求两个值之和恰好为 $k$，对于某个 $x$，只可能和 $y = k-x$ 形成一对，其他的都不可能进行配对

那么这样的 $(x, y)$ 之间进行唯一的配对，对答案的贡献数量是 $min(Cnt[x], Cnt[k - x])$

注意，当 $x = y$ 时，贡献的对数只有 $Cnt[x]/2$

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18, N = 2e5 + 5;
int n, k, a[N], b[N], C;
void solve(){
    cin >> n >> k;
    map<int, int> Cnt;
    for(int i = 1; i <= n; ++ i)
        cin >> a[i], Cnt[a[i]] ++;
    int ans = 0;
    for(auto &[x, c] : Cnt) {
        int t = min(c, Cnt[k - x]);
        Cnt[k-x] -= t, c -= t;
        if(k == 2 * x) {
            ans += t/2;
        } else ans += t;
    }
    cout << ans << '\n';
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```
