# 2026江西八校集训联盟入门赛第二场

## A

思路：

找到最大的出现次数，然后枚举值

第一个出现最大次数的值就是答案了

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

int n, k;
void solve(){
    cin >> n >> k;
    vt(int) Cnt(k + 1);
    for(int i = 1, x; i <= n; ++ i)
        cin >> x, Cnt[x] ++;
    int mxc = *max_element(all(Cnt));
    for(int i = 1; i <= k; ++ i) {
        if(mxc == Cnt[i]) {
            cout << i << ' ' << mxc << '\n';
            return;
        }
    }
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## B 

思路：

判断 $NAME-LNN$ 这种字符串，满足如下条件则为合法字符串：

1. 只有一个 `-` 字符
2. $NAME$ 子串的长度在 $1~10$ 之间，且每个字母都必须是大写字母
3. $LNN$ 子串长度恒为 $3$，$L$ 必须为大写字母，$NN$ 必须恰好为两个数字

统计总数量就是 $n$ 个中有几个合法的

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

int n, k;
int check(const string &s){
    if(count(all(s), '-') != 1)
        return 0;
    string A = "", B = "";
    for(int i = 0, flg = 0; i < s.size(); ++ i) {
        if(s[i] == '-') {
            flg = 1;
            continue;
        }
        if(!flg) {
            A.push_back(s[i]);
        }else B.push_back(s[i]);
    }

    if(A.size() < 1 || A.size() > 10 || B.size() != 3)
        return 0;
    
    for(auto ch : A) {
        if(ch >= 'A' && ch <= 'Z') 
            continue;
        return 0;
    }
    
    if(isupper(B[0]) && isdigit(B[1]) && isdigit(B[2])) return 1;
    return 0;
}

void solve(){
    cin >> n;
    int ans = 0;
    for(int i = 0; i < n; ++ i) {
        string s;
        cin >> s;
        if(check(s)) 
            ++ ans;
    }
    Out(ans);   
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## C

思路：

类似上场 $C$ 的一个模拟题，

维护当前位置和速度，以及过程中遇到的最大位置和最小位置

每一种操作之前都先按照 $v$ 的速度走一段 $now$，再更新

在 `op=P` 时判定最大和最小位置，其他操作就是更新 $v$

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int n, x;
void solve(){
    cin >> n;
    int mx = -INF, mn = INF, now = 0, v = 0;
    for(int i = 0, x; i < n; ++ i) {
        char op;
        cin >> op;
        if(op == 'A') {
            cin >> x;
            now += v;
            v += x;
        }else if(op == 'B') {
            cin >> x;
            now += v;
            v -= x;
        }else {
            now += v;
            mn = min(mn, now);
            mx = max(mx, now);
        }
    }
    Out(now);
    Out(mx);
    Out(mn);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## D

思路：

按照题意，需要一个可以加入元素和删除最后一个元素的结构，栈就可以，在 $c++$ 中能直接用 $vector$ 定义栈，因为要求的操作 $vector$ 完全可以胜任

然后维护当前存在的所有元素，以及当前的总和，按照操作模拟就好了

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int n, x;
void solve(){
    cin >> n;
    vt(int) todo;
    int now = 0;
    for(int i = 0, op, x; i < n; ++ i) {
        cin >> op;
        if(op == 1) {
            cin >> x;
            todo.push_back(x);
            now += x;
        }else if(op == 2) {
            now -= todo.back();
            todo.pop_back();
        }else {
            Out(now);
        }
    }
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## E

思路：

注意到，数量的值域为 $10^3$，那么直接枚举其中两个的使用数量，这种枚举可以将所有方案考虑到

判定剩下的钱都买另一个是不是符合条件，合法时就是一种方案，

复杂度是 $10^6$，$50 ms$ 可以接受

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int n, a, b, c, p, q, r;
void solve(){
    cin >> n >> a >> b >> c >> p >> q >> r;
    int cnt = 0;
    for(int ia = 0; ia <= p; ++ ia) {
        for(int ib = 0; ib <= q; ++ ib) {
            int cost = ia * a + ib * b, rem = n - cost, ic = rem / c;
            if(rem >= 0 && rem % c == 0 && ic <= r) {
                ++ cnt;
            }
        }
    }
    Out(cnt);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    // cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```

## F

思路：

设一组由 $(x, y)$ 组成，要求 $x + y \le K$，问最多能产生多少组

考虑对于一个小的数字，用一个比较大的数字肯定不劣，

即对于 $x$，去匹配 $y \le K-x$ 的最大的 $y$，是最优方案，

那么排序后用双指针就能 $O(nlog{n} + n)$实现这个事情

参考代码：

```cpp
#include <bits/stdc++.h>
#define int long long
#define all(v) v.begin(), v.end()
#define mkr(a, b) make_pair(a, b)
#define dbg(x) cerr << #x << " : " << x << endl
#define vt(tp) vector<tp>
using namespace std;

using ll = long long;
using lll = __int128;
using pii = pair<int, int>;

template<typename T>
void Out(T x){cout << x << '\n';}
void Out(vector<int> &V) {for(int i = 0; i < V.size(); ++ i) cout << V[i] << " \n"[i + 1 == V.size()];}

const int INF = 4e18;
int n, K;
void solve(){
    cin >> n >> K;
    vt(int) a(n + 1);
    for(int i = 1; i <= n; ++ i)
        cin >> a[i];
    sort(a.begin() + 1, a.end());
    int ans = 0, i = 1, j = n;
    while(i < j) {
        while(j > i && a[j] + a[i] > K) 
            -- j;
        if(j - i >= 1 && a[i] + a[j] <= K){
            ++ ans;
            -- j;
            ++ i;
        }
    }
    Out(ans);
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);

    int t = 1;
    cin >> t;
    while (t--) {
        solve();
    }

    return 0;
}
```
